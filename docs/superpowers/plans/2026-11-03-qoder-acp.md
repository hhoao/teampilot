# Qoder CLI (ACP-only) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `qodercli` as a first-class ACP-only TeamPilot CLI with provider-isolated login, full config surface, and mixed-team ACP seats.

**Architecture:** Mirror `services/cli/codebuddy/`: one `CliToolDefinition` whose capabilities write the CLI's Claude-compatible config (`QODER_CONFIG_DIR` → `settings.json`, `mcp.json`, `skills/`, `plugins/`, `AGENTS.md`). The one real difference is credentials: Qoder's `QODER_CONFIG_DIR` also isolates `.auth`, so every session config dir must be seeded with `user` + `machine_id` copied from a per-provider login.

**Tech Stack:** Flutter/Dart, `acpd` ACP client, npm installer capability, Alacritty login dialog.

## Global Constraints

- Never run `flutter test`; always `cd client && dart run tool/run_tests.dart <paths>`.
- Before claiming done: `cd client && flutter analyze --no-fatal-infos --no-fatal-warnings && dart run tool/run_tests.dart`.
- Capability-specific code lives only under `client/lib/services/cli/qoder/`; `registry/capabilities/` holds interfaces/shared implementations only.
- No `if (cli == qoder)` in `AcpSessionRuntime` / `AcpChatRuntimeCubit` / `SessionLaunchService`; behavior is capability-driven.
- All user-visible strings go to `client/lib/l10n/app_en.arb` + `app_zh.arb`.
- `QODER_CONFIG_DIR` is always the per-session/member tool dir; never write `~/.qoder`.
- ACP argv is exactly `[qodercli, --acp]`; no PTY transport is registered.
- Spec: `docs/superpowers/specs/2026-11-03-qoder-acp-design.md`.

import 'dart:convert';

import 'package:acpd/acpd.dart';

import '../../../acp/acp_agent_translator.dart';
import '../../registry/capabilities/acp_capability.dart';

/// Qoder CLI ACP quirks.
///
/// The CLI speaks ACP natively (`qodercli --acp`). The flag is hidden from
/// `--help`, so an upstream removal surfaces as "no initialize response" and
/// the runtime marks the CLI unsupported.
final class QoderAcpTranslator extends AcpAgentTranslator {
  const QoderAcpTranslator();

  @override
  List<String> get flagCandidates => const ['--acp'];

  @override
  List<String> get blockingClientMethods => const [];

  @override
  List<String> get notificationClientMethods => const [];

  @override
  String translateStopError(Object error, {required bool suppressAbort}) {
    if (error is RpcError) {
      if (error.code == ErrorCode.authRequired.code) {
        return acpStopAuthRequired;
      }
      return 'ACP error ${error.code}: ${error.message}';
    }
    return error.toString();
  }

  /// `session/new` reports an unseeded config dir as `-32000
  /// Authentication required`; surface it as the app's sign-in sentinel.
  @override
  String? translateStopMeta(StopReason reason, Map<String, Object?> meta) {
    final raw = meta['qoder.ai/errorMessage']?.toString() ?? '';
    if (raw.isEmpty) return null;
    final lower = raw.toLowerCase();
    if (raw.contains('-32000') ||
        lower.contains('auth') ||
        lower.contains('login')) {
      return acpStopAuthRequired;
    }
    try {
      final decoded = jsonDecode(raw);
      if (decoded is Map && decoded['message'] is String) {
        return (decoded['message'] as String).trim();
      }
    } on FormatException {
      // Not a JSON envelope.
    }
    return raw;
  }
}

/// Qoder launch/env declaration.
///
/// `QODER_CONFIG_DIR` isolates settings **and** `.auth`. The provider
/// capability seeds `.auth` into `sessionToolDir`; without it `session/new`
/// fails with auth-required, which is why no `authenticate` call is made.
final class QoderAcpCapability implements AcpCapability {
  const QoderAcpCapability();

  static const configDirEnvKey = 'QODER_CONFIG_DIR';

  @override
  String agentExecutable(String cliExecutable) => cliExecutable;

  @override
  List<List<String>> launchArgCandidates(String executable) => [
    [executable, '--acp'],
  ];

  @override
  AcpAgentTranslator get translator => const QoderAcpTranslator();

  @override
  String? get authenticateMethodId => null;

  @override
  Future<String?> resolveAuthenticateMethodId(Map<String, String> env) async =>
      authenticateMethodId;

  @override
  bool get forwardSessionNewMcpServers => false;

  @override
  Map<String, String> get agentStatusEnvironment => const {};

  @override
  Map<String, Object?> sessionNewMeta(Map<String, String> env) => const {};

  @override
  Map<String, String> buildEnvironment({
    required String sessionToolDir,
    required String workingDirectory,
    Map<String, String> baseEnv = const {},
  }) {
    final env = Map<String, String>.of(baseEnv);
    env[configDirEnvKey] = sessionToolDir;
    return env;
  }
}

// client/lib/services/cli/qoder/provider/qoder_auth_store.dart
import 'dart:io';

import 'package:path/path.dart' as p;

import '../../../io/filesystem.dart';

/// Qoder credential isolation.
///
/// `qodercli` keeps its login inside `<QODER_CONFIG_DIR>/.auth`
/// (`user` + `machine_id`), so a fresh session config dir cannot
/// authenticate. TeamPilot runs the login with an isolated HOME/config and
/// copies both files into a per-provider store; each session config dir is
/// seeded from that store (or from `~/.qoder` when no provider account
/// exists).
class QoderAuthStore {
  QoderAuthStore({
    required this.fs,
    required this.basePath,
    required this.pathContext,
    String? machineHome,
  }) : machineHome = machineHome ??
            (Platform.environment['HOME'] ??
                Platform.environment['USERPROFILE'] ??
                '');

  final Filesystem fs;
  final String basePath;
  final p.Context pathContext;
  final String machineHome;

  static const configDirEnvKey = 'QODER_CONFIG_DIR';
  static const userFileName = 'user';
  static const machineIdFileName = 'machine_id';

  String loginRoot(String providerId) =>
      pathContext.join(basePath, 'cli-login', 'qoder', _safe(providerId));

  String loginHome(String providerId) =>
      pathContext.join(loginRoot(providerId), 'home');
  String loginConfigDir(String providerId) =>
      pathContext.join(loginRoot(providerId), 'config');
  String loginWorkspace(String providerId) =>
      pathContext.join(loginRoot(providerId), 'workspace');

  String providerAuthDir(String providerId) => pathContext.join(
    basePath,
    'providers',
    'qoder',
    _safe(providerId),
  );

  Future<bool> hasProviderAuth(String providerId) async {
    final dir = providerAuthDir(providerId);
    return (await fs.stat(pathContext.join(dir, userFileName))).isFile &&
        (await fs.stat(pathContext.join(dir, machineIdFileName))).isFile;
  }

  /// Copies the isolated login's `.auth/{user,machine_id}` into the provider
  /// store. Returns false when the login did not complete.
  Future<bool> writeProviderAuthFromLogin(
    String providerId, {
    required String loginConfigDir,
  }) async {
    final source = pathContext.join(loginConfigDir, '.auth');
    final target = providerAuthDir(providerId);
    for (final name in const [userFileName, machineIdFileName]) {
      if (!(await fs.stat(pathContext.join(source, name))).isFile) return false;
    }
    await fs.ensureDir(target);
    for (final name in const [userFileName, machineIdFileName]) {
      final dst = pathContext.join(target, name);
      await fs.copyFile(pathContext.join(source, name), dst);
      _restrict(dst);
    }
    return true;
  }

  /// Seeds `<sessionConfigDir>/.auth` so `session/new` can authenticate.
  Future<bool> seedSessionAuth({
    required String providerId,
    required String sessionConfigDir,
  }) async {
    final source = await hasProviderAuth(providerId)
        ? providerAuthDir(providerId)
        : pathContext.join(machineHome, '.qoder', '.auth');
    if (!(await fs.stat(pathContext.join(source, userFileName))).isFile) {
      return false;
    }
    final target = pathContext.join(sessionConfigDir, '.auth');
    await fs.ensureDir(target);
    for (final name in const [userFileName, machineIdFileName]) {
      final src = pathContext.join(source, name);
      if (!(await fs.stat(src)).isFile) continue;
      final dst = pathContext.join(target, name);
      await fs.copyFile(src, dst);
      _restrict(dst);
    }
    return true;
  }

  void _restrict(String path) {
    if (Platform.isWindows) return;
    try {
      if (!File(path).existsSync()) return;
      Process.runSync('chmod', ['600', path]);
    } on Object {
      // Best effort.
    }
  }

  static String _safe(String value) {
    final safe = value.replaceAll(RegExp(r'[^A-Za-z0-9._-]'), '_');
    return safe.isEmpty ? 'provider' : safe;
  }
}

// client/test/integration/qoder_acp_local_smoke_test.dart
@Tags(['integration'])
library;

import 'dart:async';
import 'dart:io';

import 'package:flutter_test/flutter_test.dart';
import 'package:path/path.dart' as p;
import 'package:teampilot/cubits/session/acp/acp_launch_wiring.dart';

void main() {
  test('qodercli --acp completes initialize / session/new / prompt', () async {
    final executable = await _which('qodercli');
    final home = Platform.environment['HOME'] ?? '';
    final auth = p.join(home, '.qoder', '.auth');
    if (executable == null || !Directory(auth).existsSync()) {
      // ignore: avoid_print
      print('[qoder-acp] skipped: qodercli or ~/.qoder/.auth is missing');
      return;
    }

    final temp = Directory.systemTemp.createTempSync('qoder-acp-');
    Directory(p.join(temp.path, '.auth')).createSync(recursive: true);
    for (final name in const ['user', 'machine_id']) {
      final source = File(p.join(auth, name));
      if (source.existsSync()) source.copySync(p.join(temp.path, '.auth', name));
    }

    final registry = CliToolRegistry()..register(const QoderCliTool());
    final logger = AcpFrameLogger(mirrorToLog: false);
    final runtime = AcpSessionRuntime(
      spawner: productionAcpSpawner(
        registry: registry,
        tool: CliTool.qoder,
        workingDirectory: temp.path,
        logger: logger,
        handshakeTimeout: const Duration(seconds: 60),
      ),
      logger: logger,
      launchPlanFor: acpLaunchPlanResolver(registry),
    );

    try {
      await runtime.start(
        tool: CliTool.qoder,
        executable: executable!,
        sessionToolDir: temp.path,
        workingDirectory: temp.path,
        baseEnv: {'PATH': Platform.environment['PATH'] ?? ''},
      );
      expect(runtime.isStarted, isTrue);

      final reply = StringBuffer();
      final updates = runtime.updates.listen((update) {
        if (update is AcpMessageChunk && update.role == AcpRole.assistant) {
          reply.write(update.content);
        }
      });
      await runtime.prompt('Reply with exactly: pong');
      await updates.cancel();
      expect(reply.toString().toLowerCase(), contains('pong'));
    } finally {
      await runtime.close();
      try {
        temp.deleteSync(recursive: true);
      } on FileSystemException {
        // Best effort temp cleanup.
      }
    }
  }, timeout: const Timeout(Duration(minutes: 3)));
}

Future<String?> _which(String name) async {
  try {
    final result = await Process.run('which', [name]);
    if (result.exitCode != 0) return null;
    final path = (result.stdout as String? ?? '').trim();
    return path.isEmpty ? null : path;
  } on ProcessException {
    return null;
  }
}
